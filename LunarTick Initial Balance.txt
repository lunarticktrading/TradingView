request_security_no_repaint(timeframe, expression) =>
    request.security(symbol = syminfo.tickerid, timeframe = timeframe, expression = expression[barstate.isrealtime ? 1 : 0], gaps = barmerge.gaps_off)[barstate.isrealtime ? 0 : 1] // PineCoders best practice.

build_initial_balance_session(string session, int initial_balance_duration_minutes) =>
    string initial_balance_session = na
    int start_time = int(str.tonumber(str.substring(session, 0, 4)))
    //log.info("START_TIME: [" + str.tostring(start_time) + "]")
    int end_time = LunarTick.time_add_minutes(start_time, initial_balance_duration_minutes)
    //log.info("END_TIME: [" + str.tostring(end_time) + "]")
    initial_balance_session := LunarTick.pad_leading_zeroes(start_time, 4) + "-" + LunarTick.pad_leading_zeroes(end_time, 4)
    //log.info("IB_SESSION: [" + initial_balance_session + "]")
    initial_balance_session

get_label_style(LabelStyle lbl_style) =>
    switch lbl_style
        LabelStyle.none => label.style_label_left
        LabelStyle.label_up => label.style_label_up
        LabelStyle.label_down => label.style_label_down
        LabelStyle.label_left => label.style_label_left
        LabelStyle.label_right => label.style_label_right
        LabelStyle.label_lower_left => label.style_label_lower_left
        LabelStyle.label_lower_right => label.style_label_lower_right
        LabelStyle.label_upper_left => label.style_label_upper_left
        LabelStyle.label_upper_right => label.style_label_upper_right
        LabelStyle.label_center => label.style_label_center
        LabelStyle.text_outline => label.style_text_outline
        => label.style_none

get_label_textalign(LabelStyle lbl_style) =>
    switch lbl_style
        LabelStyle.none => text.align_left
        LabelStyle.label_left => text.align_left
        LabelStyle.label_right => text.align_right
        LabelStyle.label_lower_left => text.align_left
        LabelStyle.label_lower_right => text.align_right
        LabelStyle.label_upper_left => text.align_left
        LabelStyle.label_upper_right => text.align_right
        => text.align_center

get_label_bg_color(LabelStyle lbl_style, color lbl_color) =>
    switch lbl_style
        LabelStyle.none => na
        => color.new(lbl_color, i_label_bg_transparency)

get_label_text_color(LabelStyle lbl_style, color lbl_color) =>
    switch lbl_style
        LabelStyle.none => lbl_color
        => i_label_text_color

get_label_size(LabelSize lbl_size) =>
    switch lbl_size
        LabelSize.auto => size.auto
        LabelSize.tiny => size.tiny
        LabelSize.small => size.small
        LabelSize.normal => size.normal
        LabelSize.large => size.large
        LabelSize.huge => size.huge
        => size.auto

get_line_style(LineStyle value) =>
    switch value
        LineStyle.dashed => line.style_dashed
        LineStyle.dotted => line.style_dotted
        => line.style_solid

get_end_bar_index() =>
    int end_bar_index = bar_index
    for i = 1 to 499
        if na(time("", i_session, i_tz, -i))
            break
        end_bar_index := end_bar_index + 1
    end_bar_index

method initialize(InitialBalance self, int region_start_bar_index, float region_high, float region_low) =>
    self.region_start_bar_index := region_start_bar_index
    self.region_high := region_high
    self.region_low := region_low
    self.region_closed := false
    if i_highlight_initial_balance_region
        self.region_box := box.new(region_start_bar_index, region_high, region_start_bar_index, region_low, na, 0, line.style_solid, extend.none, xloc.bar_index, i_initial_balance_region_color)

method update_levels(InitialBalance self, int region_end_bar_index, float region_high, float region_low) =>
    self.region_end_bar_index := region_end_bar_index
    self.region_high := math.max(region_high, self.region_high)
    self.region_low := math.min(region_low, self.region_low)
    if not na(self.range_box)
        self.range_box.set_right(region_end_bar_index)
        self.range_box.set_top(self.region_high)
        self.range_box.set_bottom(self.region_low)
    if not na(self.region_box)
        self.region_box.set_right(region_end_bar_index)
        self.region_box.set_top(self.region_high)
        self.region_box.set_bottom(self.region_low)

        self.range_end_bar_index := right_bar_index
        if i_highlight_initial_balance_range

        range_mid = (self.region_high + self.region_low) / 2.0
        self.high_line := line.new(self.region_start_bar_index, self.region_high, right_bar_index, self.region_high, xloc.bar_index, extend.none, i_initial_balance_h_color, get_line_style(i_initial_balance_h_style), i_initial_balance_h_width)

var initial_balances_list = array.new<InitialBalance>(i_show_num_days)

var string initial_balance_session = build_initial_balance_session(i_session, INITIAL_BALANCE_DURATION)
var bool show_initial_balance = (timeframe.in_seconds() <= (INITIAL_BALANCE_DURATION * 60)) and (((INITIAL_BALANCE_DURATION * 60) % timeframe.in_seconds()) == 0)
varip int cached_range_end_bar_index = na

if show_initial_balance
    // Check for entering Initial Balance
    if na(time("", initial_balance_session, i_tz, 1)) and not na(time("", initial_balance_session, i_tz))
        // Just entered Initial Balance
        new_initial_balance = InitialBalance.new()
        new_initial_balance.initialize(bar_index, high, low)
        dead_initial_balance = initial_balances_list.shift()
        initial_balances_list.push(new_initial_balance)
        if not na(dead_initial_balance)
            dead_initial_balance.cleanup()
    else if not na(time("", initial_balance_session, i_tz))
        // In Initial Balance
        current_initial_balance = initial_balances_list.get(-1)
        if not na(current_initial_balance)
            current_initial_balance.update_levels(bar_index, high, low)
    else if not na(time("", initial_balance_session, i_tz, 1)) and na(time("", initial_balance_session, i_tz))
        // Just left Initial Balance
        if barstate.isnew
            cached_range_end_bar_index := get_end_bar_index()
        current_initial_balance = initial_balances_list.get(-1)
        if not na(current_initial_balance)
            current_initial_balance.confirm_levels(cached_range_end_bar_index)
    else if not na(time("", i_session, i_tz))
        // Outside Initial Balance but still inside Session
        if barstate.isnew
            cached_range_end_bar_index := get_end_bar_index()
        current_initial_balance = initial_balances_list.get(-1)
        if not na(current_initial_balance)
            current_initial_balance.extend(cached_range_end_bar_index)
    else if not na(time("", i_session, i_tz, 1)) and na(time("", i_session, i_tz))
        // Just left Session
        current_initial_balance = initial_balances_list.get(-1)
        if not na(current_initial_balance) and i_show_labels
            current_initial_balance.cleanup_labels()
