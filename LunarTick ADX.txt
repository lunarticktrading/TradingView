//@version=6
indicator("LunarTick ADX", overlay = false)

////////////////////////////////////////////////////////////////////////////////////////////////////
// Version History
////////////////////////////////////////////////////////////////////////////////////////////////////
// v1.0.0
// - Initial release
////////////////////////////////////////////////////////////////////////////////////////////////////


////////////////////////////////////////////////////////////////////////////////////////////////////
// Constants
////////////////////////////////////////////////////////////////////////////////////////////////////

var DISPLAY_GROUP = "DISPLAY"
var DEVELOPER_GROUP = "DEVELOPER"


////////////////////////////////////////////////////////////////////////////////////////////////////
// Enums
////////////////////////////////////////////////////////////////////////////////////////////////////

enum VersionEnum
    version = "v1.0.0"
    

////////////////////////////////////////////////////////////////////////////////////////////////////
// Inputs
////////////////////////////////////////////////////////////////////////////////////////////////////

i_adx_smoothing = input.int(14, "ADX Smoothing", minval = 1)
i_di_length = input.int(14, "DI Length", minval = 1)
i_threshold = input.float(20.0, "Threshold", minval = 0.0, step = 0.1)

// Display
i_show_di_crossovers = input.bool(true, "Show DI Crossovers", group = DISPLAY_GROUP)

// Developer
VersionEnum i_version = input.enum(VersionEnum.version, "Version", display = display.none, group = DEVELOPER_GROUP)


////////////////////////////////////////////////////////////////////////////////////////////////////
// Calculations
////////////////////////////////////////////////////////////////////////////////////////////////////

[diplus, diminus, adx] = ta.dmi(i_di_length, i_adx_smoothing)

bullish_di_cross = ta.crossover(diplus, diminus)
bearish_di_cross = ta.crossover(diminus, diplus)


////////////////////////////////////////////////////////////////////////////////////////////////////
// Plots
////////////////////////////////////////////////////////////////////////////////////////////////////

plotshape(bullish_di_cross, "Bullish DI Crossover", shape.xcross, location.belowbar, color.green, size = size.tiny, force_overlay = true, display = i_show_di_crossovers ? display.pane : display.none)
plotshape(bearish_di_cross, "Bearish DI Crossover", shape.xcross, location.abovebar, color.red, size = size.tiny, force_overlay = true, display = i_show_di_crossovers ? display.pane : display.none)

plot(adx, "ADX", color.blue, 1)
plot(diplus, "+DI", color.green, 1)
plot(diminus, "-DI", color.red, 1)
plot(i_threshold, "Threshold", color.gray, 1, display = display.pane)

bullish_cond = diplus > diminus and diplus > i_threshold
bearish_cond = diminus > diplus and diminus > i_threshold

bg_color = bullish_cond ? color.new(color.green, 90) : (bearish_cond ? color.new(color.red, 90) : na)
bgcolor(bg_color)
