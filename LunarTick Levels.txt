can_show_timeframe(string tf) =>
    timeframe.in_seconds(tf) >= timeframe.in_seconds()

get_label_style(LabelStyle lbl_style) =>
    switch lbl_style
        LabelStyle.none => label.style_none
        LabelStyle.label_up => label.style_label_up
        LabelStyle.label_down => label.style_label_down
        LabelStyle.label_left => label.style_label_left
        LabelStyle.label_right => label.style_label_right
        LabelStyle.label_lower_left => label.style_label_lower_left
        LabelStyle.label_lower_right => label.style_label_lower_right
        LabelStyle.label_upper_left => label.style_label_upper_left
        LabelStyle.label_upper_right => label.style_label_upper_right
        LabelStyle.label_center => label.style_label_center
        LabelStyle.text_outline => label.style_text_outline
        => label.style_none

get_label_textalign(LabelStyle lbl_style) =>
    switch lbl_style
        LabelStyle.label_left => text.align_left
        LabelStyle.label_right => text.align_right
        LabelStyle.label_lower_left => text.align_left
        LabelStyle.label_lower_right => text.align_right
        LabelStyle.label_upper_left => text.align_left
        LabelStyle.label_upper_right => text.align_right
        => text.align_center

get_label_size(LabelSize lbl_size) =>
    switch lbl_size
        LabelSize.auto => size.auto
        LabelSize.tiny => size.tiny
        LabelSize.small => size.small
        LabelSize.normal => size.normal
        LabelSize.large => size.large
        LabelSize.huge => size.huge
        => size.auto

get_line_style(LineStyle line_style) =>
    switch line_style
        LineStyle.solid => line.style_solid
        LineStyle.dashed => line.style_dashed
        LineStyle.dotted => line.style_dotted
        => line.style_solid

update_line_and_label(bool enabled, line ln, label lbl, int line_start_bar_index, string line_style, int line_width, string line_extend, float value, string name, color color, string tooltip) =>
    line result_ln = ln
    label result_lbl = lbl
    if enabled
        // Line
        string result_ln_extend = (bar_index == line_start_bar_index) ? extend.none : line_extend
        if na(result_ln)
            result_ln := line.new(line_start_bar_index, value, bar_index, value, xloc.bar_index, result_ln_extend, color, line_style, line_width)
        if not na(result_ln)
            line.set_xy1(result_ln, line_start_bar_index, value)
            line.set_xy2(result_ln, bar_index, value)
            line.set_extend(result_ln, result_ln_extend)
        // Label (optional)
        if i_show_labels
            string result_lbl_text = str.tostring(value, format.mintick) + " " + name
            if na(result_lbl)
                result_lbl := label.new(bar_index + i_label_offset, value, result_lbl_text, xloc.bar_index, yloc.price, color.new(color, i_label_bg_transparency), get_label_style(i_label_style), i_label_text_color, get_label_size(i_label_size), get_label_textalign(i_label_style), tooltip)
            if not na(result_lbl)
