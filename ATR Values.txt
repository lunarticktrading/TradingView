////////////////////////////////////////////////////////////////////////////////////////////////////
// Initialization
////////////////////////////////////////////////////////////////////////////////////////////////////

var bool initialized = false
var string price_format = format.mintick

once not initialized
    price_format := "0"
    string mintick_str = str.tostring(syminfo.mintick)
    int period_pos = str.pos(mintick_str, ".")
    if not na(period_pos)
        price_format := price_format + "."
        int decimal_place = (str.length(mintick_str) - period_pos) - 1
        for i = 1 to decimal_place by 1
            price_format := price_format + "#"
    //log.info("PRICE FORMAT: " + price_format)
    initialized := true


////////////////////////////////////////////////////////////////////////////////////////////////////
// Functions
////////////////////////////////////////////////////////////////////////////////////////////////////

get_tbl_rows() => ((i_mult1_enabled ? 1 : 0) + (i_mult2_enabled ? 1 : 0) + (i_mult3_enabled ? 1 : 0) + (i_mult4_enabled ? 1 : 0))

get_cell_alignment(Position pos) =>
    switch pos
        Position.top_right => text.align_right
        Position.middle_right => text.align_right
        Position.bottom_right => text.align_right
        Position.top_left => text.align_left
        Position.middle_left => text.align_left
        Position.bottom_left => text.align_left
        Position.top_center => text.align_center
        Position.middle_center => text.align_center
        Position.bottom_center => text.align_center
        => text.align_right

get_points_format() => (i_round_points_to_mintick ? format.mintick : price_format)


////////////////////////////////////////////////////////////////////////////////////////////////////
// Calculations
////////////////////////////////////////////////////////////////////////////////////////////////////

atr = ta.atr(i_atr_length)

get_atr_value_points(int bars_ago, float multiplier) => (multiplier * atr[bars_ago])

get_atr_value_ticks(int bars_ago, float multiplier) => (get_atr_value_points(bars_ago, multiplier) / syminfo.mintick)
