indicator("LunarTick Trend", overlay = true, behind_chart = true, explicit_plot_zorder = true)

////////////////////////////////////////////////////////////////////////////////////////////////////
// Imports
////////////////////////////////////////////////////////////////////////////////////////////////////

import gambcl/LunarTick/1


////////////////////////////////////////////////////////////////////////////////////////////////////
// Constants
////////////////////////////////////////////////////////////////////////////////////////////////////

var DISPLAY_GROUP = "DISPLAY"
var DEVELOPER_GROUP = "DEVELOPER"


////////////////////////////////////////////////////////////////////////////////////////////////////
// Enums
////////////////////////////////////////////////////////////////////////////////////////////////////

enum MAType
    sma = "SMA"
    ema = "EMA"

enum VersionEnum
    version = "v1.1.0"
    

////////////////////////////////////////////////////////////////////////////////////////////////////
// Inputs
////////////////////////////////////////////////////////////////////////////////////////////////////

i_timeframe = input.timeframe("", "Timeframe")
i_fast_length = input.int(9, "Fast Length", minval = 1, inline = "FastMA")
i_fast_color = input.color(color.green, "", inline = "FastMA")
i_medium_length = input.int(21, "Medium Length", minval = 1, inline = "MediumMA")
i_medium_color = input.color(color.yellow, "", inline = "MediumMA")
i_show_slow_ma = input.bool(true, "Slow Length", inline = "SlowMA")
i_slow_length = input.int(50, "", minval = 1, inline = "SlowMA", active = i_show_slow_ma)
i_slow_color = input.color(color.red, "", inline = "SlowMA", active = i_show_slow_ma)
i_type = input.enum(MAType.ema, "Type")

// Display
i_bullish_cloud_color = input.color(color.green, "Bullish Cloud", group = DISPLAY_GROUP, display = display.none)
i_bearish_cloud_color = input.color(color.red, "Bearish Cloud", group = DISPLAY_GROUP, display = display.none)
i_show_retracements = input.bool(true, "Show Retracements", inline = "RETRACEMENTS", group = DISPLAY_GROUP, display = display.none)
i_retracement_color = input.color(color.yellow, "", inline = "RETRACEMENTS", group = DISPLAY_GROUP, display = display.none)
i_lines_use_cloud_color = input.bool(false, "Lines Use Cloud Color", group = DISPLAY_GROUP, display = display.none)
i_fast_transparency = input.int(0, "Fast Line Transparency", minval = 0, maxval = 100, group = DISPLAY_GROUP, display = display.none)
i_medium_transparency = input.int(0, "Medium Line Transparency", minval = 0, maxval = 100, group = DISPLAY_GROUP, display = display.none)
i_slow_transparency = input.int(0, "Slow Line Transparency", minval = 0, maxval = 100, group = DISPLAY_GROUP, display = display.none, active = i_show_slow_ma)
i_transparency = input.int(70, "Cloud Transparency", minval = 0, maxval = 100, group = DISPLAY_GROUP, display = display.none)

// Developer
VersionEnum i_version = input.enum(VersionEnum.version, "Version", display = display.none, group = DEVELOPER_GROUP)


////////////////////////////////////////////////////////////////////////////////////////////////////
// Moving-Average Cloud
////////////////////////////////////////////////////////////////////////////////////////////////////

var can_display = (timeframe.in_seconds() <= timeframe.in_seconds(i_timeframe))
[tf_sma_fast, tf_sma_medium, tf_sma_slow, tf_ema_fast, tf_ema_medium, tf_ema_slow] = request.security(str.length(i_timeframe) == 0 ? "" : ticker.standard(syminfo.tickerid), i_timeframe, [ta.sma(close, i_fast_length), ta.sma(close, i_medium_length), ta.sma(close, i_slow_length), ta.ema(close, i_fast_length), ta.ema(close, i_medium_length), ta.ema(close, i_slow_length)], barmerge.gaps_off, barmerge.lookahead_on)

ma_fast = (i_type == MAType.sma) ? tf_sma_fast : tf_ema_fast
ma_medium = (i_type == MAType.sma) ? tf_sma_medium : tf_ema_medium
ma_slow = (i_type == MAType.sma) ? tf_sma_slow : tf_ema_slow

ma_color = ((ma_fast > ma_medium)
   ? (not i_show_retracements or close >= ma_fast ? i_bullish_cloud_color : i_retracement_color)
   : (not i_show_retracements or close <= ma_fast ? i_bearish_cloud_color : i_retracement_color))

plot_ma_fast = plot(can_display ? ma_fast : na, "Fast MA", color = color.new(i_lines_use_cloud_color ? ma_color : i_fast_color, i_fast_transparency), linewidth = 2)
plot_ma_medium = plot(can_display ? ma_medium : na, "Medium MA", color = color.new(i_lines_use_cloud_color ? ma_color : i_medium_color, i_medium_transparency), linewidth = 2)
plot_ma_slow = plot(can_display ? ma_slow : na, "Slow MA", color = color.new(i_lines_use_cloud_color ? ma_color : i_slow_color, i_slow_transparency), linewidth = 2, display = (i_show_slow_ma ? display.all : display.none))
fill(plot_ma_fast, plot_ma_slow, color.new(ma_color, i_transparency), "MA Slow Cloud", display = (i_show_slow_ma ? display.all : display.none))
fill(plot_ma_fast, plot_ma_medium, color.new(ma_color, i_transparency), "MA Medium Cloud")
